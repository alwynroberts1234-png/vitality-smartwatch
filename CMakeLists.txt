cmake_minimum_required(VERSION 3.20)
option(VITALITY_HOST_TESTS "Build portable core tests without Zephyr" OFF)
set(CORE_SOURCES src/app/state.c src/assets/asset_manager.c src/sensors/simulator.c src/vitality/engine.c)
if(VITALITY_HOST_TESTS)
  project(vitality_core C)
  enable_testing()
  add_executable(core_tests tests/core_tests.c ${CORE_SOURCES})
  target_include_directories(core_tests PRIVATE include)
  target_compile_features(core_tests PRIVATE c_std_99)
  if(MSVC)
    target_compile_options(core_tests PRIVATE /W4)
  else()
    target_compile_options(core_tests PRIVATE -Wall -Wextra -Werror)
  endif()
  add_test(NAME core COMMAND core_tests)
else()
  find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
  project(vitality_watch)
  target_include_directories(app PRIVATE include)
  target_sources(app PRIVATE src/main.c src/app/event_bus.c ${CORE_SOURCES})
  target_sources_ifdef(CONFIG_VITALITY_UI app PRIVATE src/ui/ui_manager.c)
  target_sources_ifdef(CONFIG_VITALITY_QSPI_ASSETS app PRIVATE src/storage/qspi.c src/assets/lvgl_qspi_fs.c)
endif()
